# Commit-derived app versions; generated inputs stay in the build directory.
# Source archives, including an export nested in another checkout, use dev values.
set(LLCAST_SOURCE_VERSION "0.0.0-dev")
set(LLCAST_VERSION "0.0.0")
set(LLCAST_BUILD "1")
set(LLCAST_FILE_VERSION "0,0,0,0")
find_package(Git QUIET)
if(GIT_FOUND)
    execute_process(COMMAND "${GIT_EXECUTABLE}" rev-parse --show-toplevel
        WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}"
        OUTPUT_VARIABLE LLCAST_GIT_ROOT OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
    if(LLCAST_GIT_ROOT STREQUAL CMAKE_CURRENT_SOURCE_DIR)
        execute_process(COMMAND "${CMAKE_COMMAND}" -E env TZ=UTC "${GIT_EXECUTABLE}"
            log -1 "--format=%cd;%H" --date=format-local:%Y.%m.%d.%H%M
            WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}"
            OUTPUT_VARIABLE LLCAST_COMMIT OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
        if(LLCAST_COMMIT MATCHES "^([0-9]+)\\.([0-9]+)\\.([0-9]+)\\.([0-9]+);([0-9a-f]+)$")
            set(LLCAST_VERSION "${CMAKE_MATCH_1}.${CMAKE_MATCH_2}.${CMAKE_MATCH_3}")
            string(SUBSTRING "${CMAKE_MATCH_5}" 0 6 LLCAST_HASH)
            set(LLCAST_SOURCE_VERSION "${LLCAST_VERSION}-${LLCAST_HASH}")
            set(LLCAST_MONTH_DAY "${CMAKE_MATCH_2}${CMAKE_MATCH_3}")
            set(LLCAST_PARTS "${CMAKE_MATCH_1};${CMAKE_MATCH_2};${CMAKE_MATCH_3};${CMAKE_MATCH_4}")
            set(LLCAST_YEAR "${CMAKE_MATCH_1}")
            set(LLCAST_NUMBERS "")
            foreach(part IN LISTS LLCAST_PARTS)
                string(REGEX REPLACE "^0+" "" part "${part}")
                if(part STREQUAL "")
                    set(part 0)
                endif()
                list(APPEND LLCAST_NUMBERS "${part}")
            endforeach()
            list(JOIN LLCAST_NUMBERS "," LLCAST_FILE_VERSION)
            string(REGEX REPLACE "^0+" "" LLCAST_MONTH_DAY "${LLCAST_MONTH_DAY}")
            list(GET LLCAST_NUMBERS 3 LLCAST_HHMM)
            set(LLCAST_BUILD "${LLCAST_YEAR}.${LLCAST_MONTH_DAY}.${LLCAST_HHMM}")
        endif()
        # A new commit updates version resources even in an existing build tree.
        execute_process(COMMAND "${GIT_EXECUTABLE}" symbolic-ref -q HEAD
            WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}"
            OUTPUT_VARIABLE LLCAST_REF OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
        foreach(ref HEAD packed-refs "${LLCAST_REF}")
            if(NOT ref STREQUAL "")
                execute_process(COMMAND "${GIT_EXECUTABLE}" rev-parse --git-path "${ref}"
                    WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}"
                    OUTPUT_VARIABLE LLCAST_REF_FILE OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
                get_filename_component(LLCAST_REF_FILE "${LLCAST_REF_FILE}" ABSOLUTE
                    BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
                if(EXISTS "${LLCAST_REF_FILE}")
                    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${LLCAST_REF_FILE}")
                endif()
            endif()
        endforeach()
    endif()
endif()
